991  rank->_dpus.push_back(dpu);
992  _dpus.push_back(dpu);
993  }

996  _ranks.push_back(rank);

1003  {
1004  std::ostream *LogStream = (std::ostream *)Arg;
1005  char *str;
1006  va_list ap;
1007  va_start(ap, Fmt);
1008  if (vasprintf(&str, Fmt, ap) == -1) {
1009  va_end(ap);
1010  return DPU_ERR_SYSTEM;
1011  }
1012 
1013  *LogStream << str;
1014 
1015  free(str);
1016  return DPU_OK;
